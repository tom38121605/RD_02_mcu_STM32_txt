

说明1： C011的优先级分组是固定的，所以不需要设置 （优先级分组是第几组待研究？）

说明2： 这里的IP[9>>2]，是因为C011中的IP寄存器是32位的，等于指南里的4个IP寄存器，可以控制4个中断的优先级  (相当于把四个中断对应的四个子IP寄存器合并到一个IP寄存器)

       子IP      总IP寄存器

        0 >>2  --  0
        1 >>2  --  0
        2 >>2  --  0
        3 >>2  --  0

        4 >>2  --  1
        5 >>2  --  1
        6 >>2  --  1
        7 >>2  --  1

        8 >>2  --  2
        9 >>2  --  2






~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ LL uart 的 dma tx的步骤 ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~



---------------- dma ch1 init  ---------------------


--- MX_GPDMA1_Init ---

01. 使能dma1的时钟clock
    RCC->AHBENR.0=1; 

02. 设置dma1 CH1的优先级为0 (最高)
    NVIC->IP[2].15-14 = priority=0;

03. 使能dma1 ch1的irq中断 
    NVIC->ISER[0] = 1<< 9;



--- MX_USART1_UART_Init 之一  --- 

11. 设置usart1的时钟，设置gpio时钟
    RCC->APBSMENR2.14 =1;    //UART1
    RCC->IOPENR.0=1;         //PA

12. 设置usart1的引脚


--- MX_USART1_UART_Init 之二  --- 

13. DMA1 CH1设置，设置优先级，按字节传输，源地址递增，byte模式


--- MX_USART1_UART_Init 之三  --- 

21. 设置uart1的优先级，使能uart1的irq总中断 
    NVIC->IP[5].31-30  =  priority=0;
    NVIC->ISER[0] = 1 << 27;

22. 设置uart1的初始化参数, 波特率2M，过采样率为16

23. 开启uart1
    CR1.0=1;  



---------------- dma ch1 tx 发送操作 ---------------------

--- LL_DMA_Start_IT --- 

30. 关闭 dma CH1
    DMA1_Channel1->CCR.0=0;

31. 设置dma ch1 传输数据长度，源地址，目标地址
    DMA1_Channel1->CNDTR = 6;
    DMA1_Channel1->CMAR = (uint32_t)aTxBuffer;  
    DMA1_Channel1->CPAR = (uint32_t)&USART1->TDR;  

32. 设置CFCR, 清0 TC和ERR中断标志
    DMA1->CPAR |= 1<<1 | 1<<3;
 
33. 使能dma ch1的TC和ERR中断标志
     DMA1_Channel1->CCR.1,3=1,1;

34. 开启dma ch1
    DMA1_Channel1->CCR.0=1;



//--- main --- 

41. uart设置，清0 TC中断标志
    USART1->ICR |= 1<<6; 

42. uart设置，使能uart的dma的传输(发送)
    USART1->CR3=1<<7; 



---------------- dma ch1 tx 中断处理---------------------


---- HAL_DMA_IRQHandler ----TC中断处理 ------

51. dma ch1清0 TC中断标志 
    DMA1->IFCR.1=1


---- HAL_DMA_IRQHandler ----ERR中断处理 ------

52. dma ch1清ERR中断标志
    DMA1->IFCR.3=1

功能： 设置dma ch1的传输数据长度，源地址，目标地址
       清0 dma ch1的tc和err中断 
       使能dma ch1的tc和err中断 
       开启dma ch1

参数：DMA1, aTxBuffer, &USART1->TDR, 6
LL_DMA_Start_IT( *dmainst, SrcAddress, DstAddress, SrcDataSize)


   DMA_Channel_TypeDef *tmp;
   tmp = (DMA_Channel_TypeDef *)(__LL_DMA_GET_CHANNEL_INSTANCE(dmainst, LL_DMA_CHANNEL_1));  
   
   
   //关闭 dma CH1
   CLEAR_BIT(tmp->CCR, 1<<0);     
     //DMA1_Channel1->CCR.0=0;   
   
   
   //传输数据长度
   MODIFY_REG(tmp->CNDTR, DMA_CNDTR_NDT, 6);
     //DMA1_Channel1->CNDTR = 6;

   //设置通道1的源数据地址
   WRITE_REG(tmp->CMAR,  (uint32_t)SrcAddress);
     //DMA1_Channel1->CMAR = (uint32_t)aTxBuffer;    
     
   //设置通道1的目标数据地址
   WRITE_REG(tmp->CPAR,  (uint32_t)DstAddress);
     //DMA1_Channel1->CPAR = (uint32_t)&USART1->TDR;       

   //清0寄存器CFCR中的tc和err中断标志               
   SET_BIT(dmainst->IFCR, 1<<1 | 1<<3);               
     //DMA1->CPAR |= 1<<1 | 1<<3;
               
   //使能寄存器CCR中的tc和err中断             
   SET_BIT(tmp->CCR, 1<<1 | 1<<3);  
     //DMA1_Channel1->CCR.1,3=1,1;
   
   //开启 dma CH1
   SET_BIT(tmp->CCR, 1<<0);  
     //DMA1_Channel1->CCR.0=1;
