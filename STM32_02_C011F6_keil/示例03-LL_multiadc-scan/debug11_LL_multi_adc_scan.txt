MX_ADC1_Init( )

    //CCIPR.31-30=00, 设置adc的时钟源位SYSCLK  (本地寄存器)
    LL_RCC_SetADCClockSource(LL_RCC_ADC_CLKSOURCE_SYSCLK);

     //APBSMENR2.20=1, 使能adc时钟   (本地寄存器)
    LL_APB2_GRP1_EnableClock(LL_APB2_GRP1_PERIPH_ADC);

    //IOPSMENR.0=1，使能PORTA的时钟
    LL_IOP_GRP1_EnableClock(LL_IOP_GRP1_PERIPH_GPIOA);


    //初始化adc引脚PA0，PA1

    GPIO_InitStruct.Pin = ADC0_Pin;
    GPIO_InitStruct.Mode = LL_GPIO_MODE_ANALOG;
    GPIO_InitStruct.Pull = LL_GPIO_PULL_NO;
    LL_GPIO_Init(ADC0_GPIO_Port, &GPIO_InitStruct);

    GPIO_InitStruct.Pin = ADC1_Pin;
    GPIO_InitStruct.Mode = LL_GPIO_MODE_ANALOG;
    GPIO_InitStruct.Pull = LL_GPIO_PULL_NO;
    LL_GPIO_Init(ADC1_GPIO_Port, &GPIO_InitStruct);


    //初始化adc参数，CFGR1.4-3,5,15=00,0,0， 12位精度，右对齐，无auto-off等； CFGR2.31-30=01，2分频 
    ADC_InitStruct.Clock = LL_ADC_CLOCK_SYNC_PCLK_DIV2;         //CFGR2.31-30=01，2分频 
    ADC_InitStruct.Resolution = LL_ADC_RESOLUTION_12B;          //CFGR1.4-3=00，12位精度
    ADC_InitStruct.DataAlignment = LL_ADC_DATA_ALIGN_RIGHT;     //CFGR1.5=0，右对齐
    ADC_InitStruct.LowPowerMode = LL_ADC_LP_MODE_NONE;          //CFGR1.15-14=00，关闭auto off，关闭wait  （最省电） 
    LL_ADC_Init(ADC1, &ADC_InitStruct);


    //CFGR1.21=1, 另一个寄存器CHSELR的4个bit表示一个通道，8个通道可一次性自动依次转换；
    LL_ADC_REG_SetSequencerConfigurable(ADC1, LL_ADC_REG_SEQ_CONFIGURABLE);

     //等待ISR.13=1, 上面的通道配置CFGR1已完成
     while (LL_ADC_IsActiveFlag_CCRDY(ADC1) == 0);

     //清 ISR.13=0
     LL_ADC_ClearFlag_CCRDY(ADC1);

    //使用software转换,使用通道SQ1-SQ2，不使能"不连续模式",单次触发模式, 不使能dma，adc保留旧数据
    ADC_REG_InitStruct.TriggerSource = LL_ADC_REG_TRIG_SOFTWARE;              // CFGR1.11-10=00, 使用software转换
    ADC_REG_InitStruct.SequencerLength = LL_ADC_REG_SEQ_SCAN_ENABLE_2RANKS;   // CHSELR.11-8=f, 把SQ3设置为f，这里的f相当于结束符，只用前面SQ1,SQ2两个通道，后面SQ3-SQ8都禁用
    ADC_REG_InitStruct.SequencerDiscont = LL_ADC_REG_SEQ_DISCONT_DISABLE;     // CFGR1.16=0, 不使能"不连续模式"  (如使能卡住连续模式, 到时单次和连续都不能触发）
    ADC_REG_InitStruct.ContinuousMode = LL_ADC_REG_CONV_SINGLE;               // CFGR1.13=0, 单次触发模式
    ADC_REG_InitStruct.DMATransfer = LL_ADC_REG_DMA_TRANSFER_NONE;            // CFGR1.0=0, 不使能dma
    ADC_REG_InitStruct.Overrun = LL_ADC_REG_OVR_DATA_PRESERVED;               // CFGR1.12=0, adc保留旧数据
    LL_ADC_REG_Init(ADC1, &ADC_REG_InitStruct);

    //CFGR2.0=0 , 不使能过采样
    LL_ADC_SetOverSamplingScope(ADC1, LL_ADC_OVS_DISABLE);

    //CFGR2.29=0, 不使能低频率触发
    LL_ADC_SetTriggerFrequencyMode(ADC1, LL_ADC_CLOCK_FREQ_MODE_HIGH);

    //SMPR.2-0=6, SMP1 是79.5 cycles
    LL_ADC_SetSamplingTimeCommonChannels(ADC1, LL_ADC_SAMPLINGTIME_COMMON_1, LL_ADC_SAMPLINGTIME_79CYCLE_5);

    //SMPR.6-4=6, SMP2 是79.5 cycles
    LL_ADC_SetSamplingTimeCommonChannels(ADC1, LL_ADC_SAMPLINGTIME_COMMON_2, LL_ADC_SAMPLINGTIME_79CYCLE_5);

    //IER.2=0, 不使能EOC转换完成中断
    LL_ADC_DisableIT_EOC(ADC1);

    //IER.3=0, 不使能EOS中断
    LL_ADC_DisableIT_EOS(ADC1);

     //CR.28=1，使能adc内部稳压器
     LL_ADC_EnableInternalRegulator(ADC1);

     //延时等待ADC内部稳压器稳定
     uint32_t wait_loop_index;
     wait_loop_index = ((LL_ADC_DELAY_INTERNAL_REGUL_STAB_US * (SystemCoreClock / (100000 * 2))) / 10);
     while(wait_loop_index != 0) 
        wait_loop_index--;
     

    //CHSELR.3-0 = 0 , 选择第0通道 
    LL_ADC_REG_SetSequencerRanks(ADC1, LL_ADC_REG_RANK_1, LL_ADC_CHANNEL_0);

    //SMPR.8=0, ADC1采用SMP1中的cycles数
    LL_ADC_SetChannelSamplingTime(ADC1, LL_ADC_CHANNEL_0, LL_ADC_SAMPLINGTIME_COMMON_1);

    //CHSELR.7-4 = 1, 选择第1通道 
    LL_ADC_REG_SetSequencerRanks(ADC1, LL_ADC_REG_RANK_2, LL_ADC_CHANNEL_1);


     //等待ISR.13=1, 上面的通道配置CFGR1已完成
     while (LL_ADC_IsActiveFlag_CCRDY(ADC1) == 0);

     //清 ISR.13=0
     LL_ADC_ClearFlag_CCRDY(ADC1);

     //SMPR.9=0, ADC1采用SMP1中的cycles数
     LL_ADC_SetChannelSamplingTime(ADC1, LL_ADC_CHANNEL_1, LL_ADC_SAMPLINGTIME_COMMON_1);

ADC_Activate( )
 

  __IO uint32_t idmaback = 0U;


  if (LL_ADC_IsEnabled(ADC1) == 0)    //如 CR.0=0, 则ADC尚未启动则如下操作   

    //CR.28=1，使能adc内部稳压器
    LL_ADC_EnableInternalRegulator(ADC1);

    //延时等待内部稳压器稳定
    wait_loop_index = ((LL_ADC_DELAY_INTERNAL_REGUL_STAB_US * (SystemCoreClock / (100000 * 2))) / 10);
    while(wait_loop_index != 0) 
      wait_loop_index--;


    //下面两句是dma相关 （可忽略）
    //idmaback = LL_ADC_REG_GetDMATransfer(ADC1);
      idmaback = CFGR1.1-0;

    //LL_ADC_REG_SetDMATransfer(ADC1, LL_ADC_REG_DMA_TRANSFER_NONE);
      CFGR1.1-0=00;
 

    //CR.31=1, 开启adc的校准
    LL_ADC_StartCalibration(ADC1);

    //等待CR.31=0, 完成adc校准 
    while (LL_ADC_IsCalibrationOnGoing(ADC1) != 0);
 
    
    //下面两小节是dma相关 （可忽略）
    //LL_ADC_REG_SetDMATransfer(ADC1, idmaback);
      CFGR1.1-0=idmaback;

    //等待dma设置稳定
    wait_loop_index = (ADC_DELAY_CALIB_ENABLE_CPU_CYCLES >> 1);
    while(wait_loop_index != 0)    
      wait_loop_index--;
        

    //CR.0=1, 使能adc
    LL_ADC_Enable(ADC1);

    //等待ISR.0=1,  adc已准备好采样
    while (LL_ADC_IsActiveFlag_ADRDY(ADC1) == 0);
