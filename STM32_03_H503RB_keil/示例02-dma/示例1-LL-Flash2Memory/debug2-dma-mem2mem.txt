
说明1：原来循环模式和非循环模式的区别，就是dma完成后，是否把两边buf的指针移到起始位的区别，并且循环模式自动启动dma的下一次传输。而非循环模式需手动启动dma的下一次传输。

说明2：半完成中断，是在自动循环模式时（circular mode），为了避免自动循环时回滚覆盖已接收的数据

说明3：对于 DMA 传输 9 个字节的情况，通常传输 4 个字节时会触发半完成中断。             （具体需根据不同单片机和使用情况验证）

说明4：SINC=0是指大地址的递增搬运，sdec=0，是指大地址内部小地址按顺序搬运

MX_GPDMA1_Init( )
 

   LL_DMA_InitTypeDef DMA_InitStruct = {0};

   //使能 GPDMA1的clk
   LL_AHB1_GRP1_EnableClock(LL_AHB1_GRP1_PERIPH_GPDMA1);

   //设置GPDMA1的优先级和使能irq
   NVIC_SetPriority(GPDMA1_Channel0_IRQn, NVIC_EncodePriority(NVIC_GetPriorityGrouping(),0, 0));
   NVIC_EnableIRQ(GPDMA1_Channel0_IRQn);


   // 1. 设置源地址和目标地址（先设为0，后面会改）
   DMA_InitStruct.SrcAddress = 0x00000000U;     // 数据从哪里来（比如数组A）
   DMA_InitStruct.DestAddress = 0x00000000U;    // 数据搬到哪里去（比如数组B）

   // 2. 设置搬运方向：从内存到内存（不是从外设到内存）
   DMA_InitStruct.Direction = LL_DMA_DIRECTION_MEMORY_TO_MEMORY;

   // 3. 设置搬运方式：一次只搬1批数据（不是连续搬多批）
   DMA_InitStruct.BlkHWRequest = LL_DMA_HWREQUEST_SINGLEBURST;

   // 4. 设置数据对齐方式：零填充（比如数据不够4字节时，后面补0）
   DMA_InitStruct.DataAlignment = LL_DMA_DATA_ALIGN_ZEROPADD;

   // 5. 设置每次搬运的数据量：每次搬1个“字”（Word，通常是4字节）
   DMA_InitStruct.SrcBurstLength = 1;
   DMA_InitStruct.DestBurstLength = 1;
   DMA_InitStruct.SrcDataWidth = LL_DMA_SRC_DATAWIDTH_WORD;        //这个好像有点特别，一般式1个字节，这里是4个字节
   DMA_InitStruct.DestDataWidth = LL_DMA_DEST_DATAWIDTH_WORD;      //这个好像有点特别，一般式1个字节，这里是4个字节


   // 6. 设置地址递增模式：每搬完1次，源地址和目标地址都往后移1个“字”
   DMA_InitStruct.SrcIncMode = LL_DMA_SRC_INCREMENT;
   DMA_InitStruct.DestIncMode = LL_DMA_DEST_INCREMENT;

   // 7. 设置优先级：低优先级（有其他更紧急的任务时先让道）
   DMA_InitStruct.Priority = LL_DMA_LOW_PRIORITY_HIGH_WEIGHT;

   // 8. 设置要搬运的数据总长度（先设为0，后面会改）
   DMA_InitStruct.BlkDataLength = 0x00000000U;

   // 9. 设置搬运模式：普通模式（搬完一次就停止，不是循环搬）
   DMA_InitStruct.Mode = LL_DMA_NORMAL;

   // 10. 设置触发模式：整批数据搬完后触发事件
   DMA_InitStruct.TriggerMode = LL_DMA_TRIGM_BLK_TRANSFER;


   // 11. 设置触发极性：不使用触发信号（数据来了就搬）
   DMA_InitStruct.TriggerPolarity = LL_DMA_TRIG_POLARITY_MASKED;

   // 12. 设置触发选择（先设为0，后面会改）
   DMA_InitStruct.TriggerSelection = 0x00000000U;

   // 13. 设置传输完成事件模式：整批数据搬完后才通知
   DMA_InitStruct.TransferEventMode = LL_DMA_TCEM_BLK_TRANSFER;

   // 14. 设置使用的通道端口：都使用端口1
   DMA_InitStruct.SrcAllocatedPort = LL_DMA_SRC_ALLOCATED_PORT1;
   DMA_InitStruct.DestAllocatedPort = LL_DMA_DEST_ALLOCATED_PORT1;
   DMA_InitStruct.LinkAllocatedPort = LL_DMA_LINK_ALLOCATED_PORT1;


   // 15. 设置链表执行模式：一次性执行完所有搬运任务
   DMA_InitStruct.LinkStepMode = LL_DMA_LSM_FULL_EXECUTION;

   // 16. 设置链表基地址和偏移量（先设为0，后面会改）
   DMA_InitStruct.LinkedListBaseAddr = 0x00000000;        //待理解
   DMA_InitStruct.LinkedListAddrOffset = 0x00000000;      //待理解


   //初始化DMA1通道0：把上面的配置应用到GPDMA1的通道0
   LL_DMA_Init(GPDMA1, LL_DMA_CHANNEL_0, &DMA_InitStruct);
