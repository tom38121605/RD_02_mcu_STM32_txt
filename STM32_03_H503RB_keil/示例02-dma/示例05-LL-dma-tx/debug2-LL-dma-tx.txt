   数据流：handle_GPDMA1_Channel1->XferCpltCallback      =  UART_DMATransmitCplt;

~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ LL uart 的 dma tx的步骤 ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~



---------------- dma ch0 init  ---------------------


--- MX_GPDMA1_Init ---

01. 使能dma1的时钟clock
02. 设置dma1 CH0的优先级为0 (最高)
03. 使能dma1 ch0的irq中断 



--- MX_USART1_UART_Init 之一  --- 

11. 设置usart1的时钟，设置gpio时钟
12. 设置usart1的引脚


--- MX_USART1_UART_Init 之二  --- 

21. DMA1 CH0设置CCR ,CRT1,CRT2, 设置优先级，按字节传输，目标地址递增，每次只搬运1个数据(不是一批数据)


--- MX_USART1_UART_Init 之三  --- 

31. 设置uart1的优先级，使能uart1的irq总中断 
32. 设置uart1的初始化参数, 波特率2M，过采样率为16
33. 开启uart1



---------------- dma ch0 tx 发送操作 ---------------------

--- LL_DMA_Start_IT --- 

31. 设置dma ch0的CCR,传输数据长度，源地址，目标地址
32. 设置CFCR, 清0所有中断标志
33. 使能dma ch0的完成中断,传输err中断，更新链接err中断，用户设置err中断，溢出中断
34. 使能dma ch0的半完成中断
35. 开启dma ch0


//--- main --- 

41. uart设置，ICR.6=1, 清0传输完成的中断标志
42. uart设置，CR3.7=1，使能uart的dma的传输(发送)



---------------- dma ch0 tx 中断处理---------------------


---- GPDMA1_Channel1_IRQHandler ---- HT中断处理 ------

51. dma ch0清HT中断标志CFCR.9
52. 调用半完成中断服务函数 UART_DMATxHalfCplt，空函数


---- HAL_DMA_IRQHandler ----TC中断处理 ------

53. dma ch0清TC和HT中断标志CFCR.8和 CFCR.9
54. UNLOCK解锁 dma ch0

 NVIC_EncodePriority (  PriorityGroup,   PreemptPriority,  SubPriority)
 
  PriorityGroupTmp =  0;  //临时分组编号为0  （需根据临时编号计算出优先级分组）

  //PreemptPriorityBits = ((7  - 0) >   __NVIC_PRIO_BITS)  ?   __NVIC_PRIO_BITS  :  (7  - 0);  //__NVIC_PRIO_BITS=4
       PreemptPriorityBits =__NVIC_PRIO_BITS =4;     //查表对应优先级分组为3，ISER中前3位抢占优先级和1位子优先级  //__STM32开发指南  P107

  //SubPriorityBits = ((PriorityGroupTmp + __NVIC_PRIO_BITS ) < 7 ) ? 0 : (PriorityGroupTmp - 7 ) + __NVIC_PRIO_BITS;
     SubPriorityBits = 0;    


   //AAA =  (  PreemptPriority &  ( (1 <<  PreemptPriorityBits ) - 1 ) ) << SubPriorityBits 
      AAA =  ( 0 &  ( 1<<4 - 1 ) ) << 0  = 0 & 15 =0;

   //BBB =  ((SubPriority  & ((1 << (SubPriorityBits )) - 1 )))
      BBB = 0 & ( 1<<0 - 1 )  = 0 & 0 =0;

  return ( AAA | BBB );  //0

       使能dma ch0的完成中断,传输err中断，更新链接err中断，用户设置err中断，溢出中断

       开启dma ch0

LL_DMA_Start_IT( *dmainst, SrcAddress, DstAddress, SrcDataSize)

   //传输数据长度
   MODIFY_REG(dmainst->CBR1, DMA_CBR1_BNDT, (TXBUFFERSIZE & DMA_CBR1_BNDT));
     //GPDMA1_Channel0->CBR1 = 6;  

   //设置通道1的源数据地址
   WRITE_REG(dmainst->CSAR, (uint32_t)SrcAddress);
     //GPDMA1_Channel0->CSAR = (uint32_t)aTxBuffer;     

   //设置通道1的目标数据地址
   WRITE_REG(dmainst->CDAR, (uint32_t)DstAddress);
     //GPDMA1_Channel0->CDAR = (uint32_t)&USART1->TDR;         

   //清0所有CFCR中的下列中断标志
   SET_BIT(dmainst->CFCR, DMA_CSR_TCF | DMA_CSR_HTF | DMA_CSR_DTEF | DMA_CSR_ULEF | DMA_CSR_USEF | DMA_CSR_SUSPF | DMA_CSR_TOF);
     //GPDMA1_Channel0->CFCR =  1<<8 |1<<9 | 1<<10 | 1<<11 | 1<<12 | 1<<13 | 1<<14; 
               
   //使能完成中断,传输err中断，更新链接err中断，用户设置err中断，溢出中断   
   SET_BIT(dmainst->CCR, DMA_CCR_TCIE | DMA_CCR_DTEIE | DMA_CCR_ULEIE | DMA_CCR_USEIE | DMA_CCR_TOIE);
      //GPDMA1_Channel0->CCR = 1<<8 | 1<<10 | 1<<11 | 1<<12 | 1<<14;                        

   //使能半完成中断
   SET_BIT(dmainst->CCR, DMA_CCR_HTIE);
      //GPDMA1_Channel0->CCR  |=1<<9;                              

   //开启 dma CH0
   LL_DMA_EnableChannel(GPDMA1, LL_DMA_CHANNEL_0);   
      //GPDMA1_Channel0->CCR  |=1<<0;
