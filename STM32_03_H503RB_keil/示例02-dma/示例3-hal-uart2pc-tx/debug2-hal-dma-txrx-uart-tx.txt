~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ hal uart 的 dma tx的步骤 ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~



---------------- dma ch0 init  ---------------------


--- MX_GPDMA1_Init --- 

01. 使能dma1的时钟clock
02. 设置dma1 CH0的优先级为0 (最高)
03. 使能dma1 ch0的irq中断 


---------------- uart init  ---------------------


--- HAL_UART_MspInit --- 

11. 使能USART1 clock，使能GPIOC clock，设置uart引脚
12. DMA1CH0设置CCR ,CRT1,CRT2, 设置优先级，目标地址递增，每次只搬运1个数据(不是一批数据)
13. uart2的抢占优先级设为0（最高），使能uart2的irq中断 


--- HAL_UART_Init --- 

21. uart设置CR1, CR2, CR3, 设置数据位8，异步模式，无校验，过采样倍数16，停止位1，无流控，波特率 
22. CR1.0=1 , 打开uart


---------------- dma ch0 tx 发送操作 ---------------------

--- HAL_DMA_Start_IT --- 

//30.  __HAL_LOCK 加锁dma ch0
31. 设置dma ch0的传输数据长度，源地址，目标地址
32. 使能dma ch0的完成中断,传输err中断，更新链接err中断，用户设置err中断，溢出中断
33. 使能dma ch0的半完成中断
34. 开启dma ch0


--- HAL_UART_Transmit_DMA --- 

41. uart设置，ICR.6=1, 清0传输完成的中断标志
42. uart设置，CR3.7=1，使能uart的dma的发送       



---------------- dma ch0 tx 中断处理---------------------


---- GPDMA1_Channel0_IRQHandler ---- HT中断处理 ------

51. dma ch0清HT中断标志CFCR.9
52. 调用半完成中断服务函数 UART_DMATxHalfCplt，空函数


---- GPDMA1_Channel0_IRQHandler ----TC中断处理 ------

53. dma ch0清TC和HT中断标志CFCR.8和 CFCR.9
//54. UNLOCK解锁 dma ch0

55. 调用完成中断服务函数 UART_DMATransmitCplt， 设置uart1的CR1.6=1, 使能uart的TC完成中断 (会接着触发普通的uart中断)    //可选，这一句和下面的uart tx中断处理也可注释



---------------- uart1 tx 中断处理---------------------


---- USART1_IRQHandler --- UART_EndTransmit_IT ----TC中断处理 ------

61. huart->TxXferCount = 0
62. uart1设置CR1.6=1,，禁止发送完成中断
63. UartReady置1  （改为led toggle）

HAL_UART_Init( *huart )
 
  huart->Lock = HAL_UNLOCKED;
 
  HAL_UART_MspInit(huart);
 

  huart->gState = HAL_UART_STATE_BUSY;
  __HAL_UART_DISABLE(huart);


  //是否设置uart的高级性能
  if (huart->AdvancedInit.AdvFeatureInit != UART_ADVFEATURE_NO_INIT) //no，不需要进行高级特性设置，跳过
    UART_AdvFeatureConfig(huart);


  //设置数据位8，异步模式，无校验，过采样倍数16，停止位1，无流控，波特率
  UART_SetConfig(huart);


  //因现在是在 asynchronous模式, 清除下面的标志

  CLEAR_BIT(huart->Instance->CR2, (USART_CR2_LINEN | USART_CR2_CLKEN));                   //LINEN=0 不配置主从模式，CLKEN=0 不能clk引脚
  CLEAR_BIT(huart->Instance->CR3, (USART_CR3_SCEN | USART_CR3_HDSEL | USART_CR3_IREN));   //SCEN=0 不使能智能卡模式，HDSEL=0 不使能半双工， IREN=0 不使能红外协议


  __HAL_UART_ENABLE(huart);     //CR1.0=1 , 打开uart
 
  return (UART_CheckIdleState(huart));

HAL_DMA_ConfigChannelAttributes( * hdma, ChannelAttributes )

  p_dma_instance = GET_DMA_INSTANCE(hdma);    
                 = GPDMA1;

  channel_idx    = 1 << (GET_DMA_CHANNEL(hdma) & 0x1F);
                 = 1 << ( ( GPDMA1_Channel1 & HAL_DMA_OFFSET_MASK) - HAL_DMA_CHANNEL_START ) / HAL_DMA_CHANNEL_SIZE 
                 = 1 << ( (0xD0 & 0x0FFF) - 0x50 ) /0x80 
                 = 1 << ( 0xD - 0x50 ) /0x80 
                 = 1 << 1

  //判断和设置给通道的特权参数
  if ((ChannelAttributes & DMA_CHANNEL_ATTR_PRIV_MASK) == DMA_CHANNEL_ATTR_PRIV_MASK)  //yes， 0x10，0x10，  
  
    if ((ChannelAttributes & DMA_CHANNEL_PRIV) == DMA_CHANNEL_PRIV)  //no，0x10，0x11     
      p_dma_instance->PRIVCFGR |= channel_idx;
     
    else                                                            //yes      
      //p_dma_instance->PRIVCFGR &= (~channel_idx);
          PRIVCFGR.1=0;                                              //关闭特权模式
