==============hI3C=======================

typedef struct __I3C_HandleTypeDef
{
  I3C_TypeDef                *Instance;                          
  I3C_InitTypeDef            Init;                      
  HAL_I3C_ModeTypeDef        Mode;        
           
  I3C_XferTypeDef            *pXferData;            
  const I3C_CCCTypeDef       *pCCCDesc;                          
  const I3C_PrivateTypeDef   *pPrivateDesc;                      

  uint32_t                   ControlXferCount;           
  uint32_t                   RxXferCount;                        
  uint32_t                   TxXferCount;                        
 
  DMA_HandleTypeDef          *hdmacr;               
  DMA_HandleTypeDef          *hdmatx;          
  DMA_HandleTypeDef          *hdmarx;         
  DMA_HandleTypeDef          *hdmasr;      

  HAL_LockTypeDef            Lock;                              
  __IO HAL_I3C_StateTypeDef  State;           
  __IO HAL_I3C_StateTypeDef  PreviousState;      
  __IO uint32_t              ErrorCode;                          
  HAL_StatusTypeDef(*XferISR)(struct __I3C_HandleTypeDef *hi3c,
                              uint32_t itFlags,
                              uint32_t itSources);                

  void(*ptrTxFunc)(struct __I3C_HandleTypeDef *hi3c);     
  void(*ptrRxFunc)(struct __I3C_HandleTypeDef *hi3c);            

} I3C_HandleTypeDef;

  

HAL_I3C_AddDescToFrame:

      hi3c->ErrorCode      = HAL_I3C_ERROR_NONE;
      hi3c->pCCCDesc       = pCCCDesc;             // null
      hi3c->pPrivateDesc   = pPrivateDesc;         // aPrivateDescriptor //私有数据结构体，分拆成下面的部分control数据和私有数据数组
      hi3c->pXferData      = pXferData;            // aContextBuffers  //control数据和要发送的私有数据数组
      hi3c->RxXferCount    = 0;
      hi3c->TxXferCount    = 0;

HAL_I3C_Ctrl_Transmit_IT:

      hi3c->State         = HAL_I3C_STATE_BUSY_TX;
      hi3c->TxXferCount   = hi3c->pXferData->TxBuf.Size;  //5
      hi3c->XferISR       = I3C_Ctrl_Tx_ISR;

      hi3c->ptrTxFunc = &I3C_TransmitWordTreatment;
